
module counter (
    input        clock,
    input  [3:0] data_in,
    input        load,
    input        up_down,
    input        resetn,
    output reg [3:0] count
);

always @(posedge clock)
begin
    if (!resetn)
        count <= 4'b0000;

    else if (load)
        count <= data_in;

    else if (up_down == 1)
    begin
        if (count == 4'd11)
            count <= 4'b0000;
        else
            count <= count + 1'b1;
    end

    else
    begin
        if (count ==4'b0000)
            count <= 4'd11;
        else
            count <= count - 1'b1;
    end
end

endmodule
